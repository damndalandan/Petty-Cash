Loading...
'; const res = await callServer('getClosedDaysForFiling'); if (!res.success || !res.data.length) { container.innerHTML = ''; if (noDaysMsg) noDaysMsg.classList.remove('hidden'); return; } if (noDaysMsg) noDaysMsg.classList.add('hidden'); const colorMap = { filed : { dot: 'bg-green-400', text: 'text-green-700', bg: 'bg-green-50 hover:bg-green-100 border-green-200', label: 'Filed' }, incomplete : { dot: 'bg-amber-400', text: 'text-amber-700', bg: 'bg-amber-50 hover:bg-amber-100 border-amber-200', label: 'Incomplete' }, never : { dot: 'bg-red-400', text: 'text-red-700', bg: 'bg-red-50 hover:bg-red-100 border-red-200', label: 'Not Filed' } }; container.innerHTML = res.data.map(d => { const c = colorMap[d.status] || colorMap.never; return ` `; }).join(''); // Auto-select first unfiled/incomplete day, else first day const firstPending = res.data.find(d => d.status !== 'filed'); const autoSelect = firstPending || res.data[0]; if (autoSelect) selectFilingDate(autoSelect.date); } // ── Select a date from the list and load its filing data ── async function selectFilingDate(date) { filingSelectedDate = date; // Highlight selected document.querySelectorAll('.filing-date-btn').forEach(btn => { btn.classList.toggle('ring-2', btn.id === 'filingDateBtn-' + date); btn.classList.toggle('ring-offset-1', btn.id === 'filingDateBtn-' + date); btn.classList.toggle('ring-amber-500', btn.id === 'filingDateBtn-' + date); }); // Update label const labelEl = document.getElementById('filingSelectedDateLabel'); if (labelEl) labelEl.textContent = date; // Enable submit button const btn = document.getElementById('submitFilingBtn'); if (btn) btn.disabled = false; // Reset checkboxes setCheckbox('chkFiling_Receipts', false); setCheckbox('chkFiling_NoReceipts', false); setCheckbox('chkFiling_Report', false); const notesEl = document.getElementById('filingNotes'); if (notesEl) notesEl.value = ''; // Hide banner const banner = document.getElementById('filingStatusBanner'); if (banner) banner.classList.add('hidden'); // Load existing record for this date if any const res = await callServer('getFilingChecklists', { from: date, to: date }); // Filter to only the exact date (getFilingChecklists now includes synthetic rows too) const rec = res.success ? res.data.find(r => r.date === date && !r.neverFiled) : null; if (rec) { setCheckbox('chkFiling_Receipts', rec.filedReceipts); setCheckbox('chkFiling_NoReceipts', rec.filedNoReceipts); setCheckbox('chkFiling_Report', rec.filedReport); if (notesEl) notesEl.value = rec.notes || ''; if (banner) { if (rec.fullyFiled) { banner.className = 'flex items-center gap-2 p-3 rounded-xl bg-green-50 border border-green-200 text-sm text-green-700 font-medium'; banner.innerHTML = '✅ Fully filed — last updated by ' + (rec.submittedBy || '—'); } else { banner.className = 'flex items-center gap-2 p-3 rounded-xl bg-amber-50 border border-amber-200 text-sm text-amber-700 font-medium'; banner.innerHTML = '⚠️ Partially filed — some items still pending'; } banner.classList.remove('hidden'); } } } function setCheckbox(id, val) { const el = document.getElementById(id); if (el) el.checked = !!val; } function renderFilingHistory(records) { const tbody = document.getElementById('filingHistoryBody'); return; } // Count unfiled = neverFiled OR incomplete const unfiled = records.filter(r => !r.fullyFiled).length; if (badge) { if (unfiled > 0) { badge.textContent = unfiled + ' unfiled'; badge.classList.remove('hidden'); } else badge.classList.add('hidden'); } // Sidebar badge const sidebarBadge = document.getElementById('filing-badge'); if (sidebarBadge) { if (unfiled > 0) { sidebarBadge.textContent = unfiled; sidebarBadge.classList.remove('hidden'); } else sidebarBadge.classList.add('hidden'); } const tick = (val) => val ? 'cursor-pointer' : ''; return ` ${r.date} ${r.neverFiled ? '—' : tick(r.filedReceipts)} ${r.neverFiled ? '—' : tick(r.filedNoReceipts)} ${r.neverFiled ? '—' : tick(r.filedReport)} ${statusBadge}${r.notes ? '
' + r.notes + '
' : ''} ${r.submittedBy || '—'} `; }).join(''); } async function submitFilingChecklist() { if (!filingSelectedDate) { toast('Please select a date to file', 'warning'); return; } const data = { date : filingSelectedDate, filedReceipts : document.getElementById('chkFiling_Receipts')?.checked || false, filedNoReceipts : document.getElementById('chkFiling_NoReceipts')?.checked || false, filedReport : document.getElementById('chkFiling_Report')?.checked || false, notes : document.getElementById('filingNotes')?.value.trim() || '' }; const btn = document.getElementById('submitFilingBtn'); btn.disabled = true; btn.textContent = 'Saving...'; const res = await callServer('saveFilingChecklist', data); btn.disabled = false; btn.innerHTML = '✓ Submit Filing Checklist'; if (res.success) { toast(res.updated ? 'Filing checklist updated ✓' : 'Filing checklist submitted ✓', 'success'); loadFilingPage(); } else { toast(res.message || 'Failed to save', 'error'); } }
